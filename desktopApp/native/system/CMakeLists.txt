cmake_minimum_required(VERSION 3.20)
project(archivetune_system LANGUAGES CXX)
find_package(JNI REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    # Static CRT: the package must not depend on vcruntime/msvcp DLLs an end user may not have.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_library(archivetune_system SHARED windows_system.cpp credential_windows.cpp)
    target_link_libraries(archivetune_system PRIVATE windowsapp shell32 ole32 user32 gdi32 dwmapi advapi32 crypt32)
    target_compile_definitions(archivetune_system PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
elseif(APPLE)
    enable_language(OBJCXX)
    add_library(archivetune_system SHARED macos_system.mm credential_macos.mm)
    target_compile_options(archivetune_system PRIVATE -fobjc-arc)
    target_link_libraries(archivetune_system PRIVATE "-framework AppKit" "-framework MediaPlayer" "-framework Foundation" "-framework Security")
else()
    message(FATAL_ERROR "Linux uses session D-Bus and system GTK via JVM bindings; no system JNI library is needed.")
endif()
target_include_directories(archivetune_system PRIVATE ${JNI_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}/../include")
