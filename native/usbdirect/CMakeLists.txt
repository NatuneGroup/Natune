cmake_minimum_required(VERSION 3.22.1)
project(natune_usbdirect_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(usbdirect_tests tests.cpp usb_direct_output.cpp)
target_include_directories(usbdirect_tests PRIVATE ..)
target_compile_options(usbdirect_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(usbdirect_tests PRIVATE pthread)
enable_testing()
add_test(NAME usbdirect_tests COMMAND usbdirect_tests)
